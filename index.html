async function pollOpenSky() { const statusText = document.getElementById("feed-status"); const pingText = document.getElementById("last-ping"); // CORS-friendly, public ADS-B query directly for IndiGo 6E 6535 const url = "https://api.adsb.lol/v2/callsign/IGO6535"; try { const res = await fetch(url); if (!res.ok) throw new Error("ADSB network response not OK"); const data = await res.json(); // Check if aircraft is currently broadcasting if (data && data.ac && data.ac.length > 0) { const plane = data.ac[0]; const lon = plane.lon; const lat = plane.lat; const altFt = plane.alt_baro === "ground" ? 0 : (plane.alt_baro || 0); const speedKts = Math.round(plane.gs || 0); const heading = Math.round(plane.track || 0); const onGround = plane.alt_baro === "ground" || altFt === 0; statusText.innerText = onGround ? "ON GROUND" : "AIRBORNE"; pingText.innerText = new Date().toLocaleTimeString(); document.getElementById("alt-val").innerText = onGround ? "On Tarmac" : `${altFt.toLocaleString()} ft`; document.getElementById("spd-val").innerText = `${speedKts} kts`; document.getElementById("hdg-val").innerText = `${heading}°`; if (lat && lon) { renderPlane(lon, lat, heading); } } else { // Flight is not currently in the sky or transponder is off statusText.innerText = "STANDBY / SCHEDULED"; pingText.innerText = "Transponder inactive"; document.getElementById("alt-val").innerText = "At Gate / Inactive"; document.getElementById("spd-val").innerText = "0 kts"; document.getElementById("hdg-val").innerText = "--"; // Keep placeholder at Kochi (COK) renderPlane(depCoords[0], depCoords[1], 42); } } catch (err) { statusText.innerText = "STANDBY"; pingText.innerText = "No active ping"; renderPlane(depCoords[0], depCoords[1], 42); console.warn("Live feed check:", err.message); } }
